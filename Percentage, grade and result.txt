 Percentage, grade and result
let marks = 420; // out of 600
let percent = marks / 6;
let grade = percent >= 75 ? "A" : percent >= 50 ? "B" : "C";
let result = percent >= 35 ? "Pass" : "Fail";
console.log("Percentage =", percent + "%");
console.log("Grade =", grade);
console.log(result);
